resources
| where type =~ 'microsoft.web/sites'
| extend fx = tostring(properties.siteConfig.linuxFxVersion),
         flex = strcat(properties.functionAppConfig.runtime.name, '|', properties.functionAppConfig.runtime.version)
| extend stack = iff(isnotempty(fx), fx, flex)
| where stack startswith 'python'
| project subscriptionId, resourceGroup, name, kind, plan = tostring(properties.sku), stack
| order by stack asc


resources
| where type in ('microsoft.web/sites', 'microsoft.web/sites/slots')
| extend fx = tostring(properties.siteConfig.linuxFxVersion),
         flex = strcat(properties.functionAppConfig.runtime.name, '|', properties.functionAppConfig.runtime.version)
| extend stack = tolower(iff(isnotempty(fx), fx, flex))
| where stack startswith 'python' or stack startswith 'powershell' or stack startswith 'docker'
| project subscriptionId, resourceGroup, name, type, kind, plan = tostring(properties.sku), stack
| order by stack asc

resources
| where type =~ 'microsoft.web/sites' and tostring(properties.siteConfig.linuxFxVersion) startswith 'python'
| project rgKey = tolower(strcat(subscriptionId, '/', resourceGroup)), app = name
| join kind=inner (resources | extend rgKey = tolower(strcat(subscriptionId, '/', resourceGroup))) on rgKey
| extend likelySameDeployment = tolower(name) == tolower(app)
| project resourceGroup, app, name, type, likelySameDeployment
| order by app asc, likelySameDeployment desc
